# 01. usr hbnvd - O(n) - O(n)
## Idea

> Lưu ý là mảng $A$ được sắp xếp tăng dần hoặc không giảm, các phần tử bị con trỏ bỏ qua thì không bao giờ cần dùng đến nữa

Ta xét $l,r$ ở các điều kiện sau:
1. $A[l] + A[r] > X$ : lúc đó dù cho ta có tăng $l$ thành $l+1, l+2, ..., r-1$ thì tổng vẫn lớn hơn $X$ vì vậy, để giảm tổng của hai số chỉ có cách giảm $r$
2. $A[l] + A[r] < X$ : lúc đó dù cho ta có giảm $r$ xuống $r-1, r-2, ..., l+1$ thì tổng vẫn nhỏ hơn $X$, vì vậy để tăng tổng của 2 số chỉ có cách tăng $l$

## CPP
```cpp
#include <bits/stdc++.h>
using namespace std;

void fileIO() {
    #ifdef LOCAL_DEBUG
        freopen("testcpp1.INP", "r", stdin);
        freopen("testcpp1.OUT", "w", stdout);
    #endif
}
#define ll long long
#define fto(i, a, b) for (int i = a; i <= b; ++i)
#define ftd(i, a, b) for (int i = a; i >= b; --i)
#define fastIO() ios_base::sync_with_stdio(false); cin.tie(nullptr)

int n, x;
int a[100005];
int main() {
    fileIO();
    fastIO();

    cin >> n >> x;
    fto(i, 1, n) cin >> a[i];

    int l = 1, r = n;
    while (a[l] + a[r] != x) {
        if (l >= r) {
            cout << "-1\n";
            return 0;
        }
        if (a[l] + a[r] > x) {
            --r;
        } else if (a[l] + a[r] < x) {
            ++l;
        }
    }
    cout << l << " " << r <<  "\n";
    
    return 0;
}
```
