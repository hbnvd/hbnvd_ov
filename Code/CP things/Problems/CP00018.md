# 01. by hbnvd - O(2\^(n\*m)) - O(n)
## Idea
- back-track qua tất cả nước đi hợp lệ

## Cpp
https://marisaoj.com/submission/1243883
%% Code chỉ ghi biến cần thiết ,core, không ghi full %%
```cpp

int n, m, mn;
int a[9][9];
// 2 mảng để tính vị trí các nước tiếp theo gọn hơn
int dx[] = {-1,  1,  2,  2,  1, -1, -2, -2};
int dy[] = {-2, -2, -1,  1,  2,  2,  1, -1};

void flying_cow(int step, int y, int x) {
    if (step == mn) {
        for (int i = 1; i <= n; ++i) {
            for (int j = 1; j <= m; ++j) {
                cout << a[i][j] << " ";
            }
            cout << "\n";
        }
        exit(0);
    }


    for (int i = 0; i < 8; ++i) {
        int ny = y+dy[i], nx = x+dx[i];
        if ( 1 <= ny && ny <= n && 1 <= nx && nx <= m && a[ny][nx] == 0 ) {
            a[ny][nx] = step+1;
            flying_cow(step+1, ny, nx);
            a[ny][nx] = 0;
        }

    }
}


void solve() {
    memset(a, 0, sizeof(a));
    cin >> n >> m;
    
    mn = m*n;
    a[1][1] = 1;
    flying_cow(1, 1, 1);
    
    return;
}
```
