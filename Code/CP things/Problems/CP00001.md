## Idea
- Chọn O , có (n cách)
	- Nếu A == C, có (n-1) cách -> B có (n-2) cách, D có (n-2) cách
	- Nếu A != C, có (n-1)P2 cách -> B có (n-3) cách, D có (n-3) cách
=> Công thức tổng quát
$$n((n-1)(n-2)(n-2)+A^2_{n-1}(n-3)(n-3))$$

## CPP
```cpp

#include <bits/stdc++.h>
using namespace std;

void fileIO() {
    #ifdef LOCAL_DEBUG
        freopen("TESTCPP.INP", "r", stdin);
        freopen("TESTCPP.OUT", "w", stdout);
    #endif
}

#define ll long long
#define oo 1000000007

int t;
ll A(ll n, int k) {
    ll ans = 1;
    for (int i = n-k+1; i <= n; ++i) {
        ans = (ans*i)%oo;
    }
    return ans;
}

int main() {
    fileIO();
    
    cin >> t;
    for (int i = 1; i <= t; ++i) {
        ll n; cin >> n;
        if (n < 3) {
            cout << "0\n";
            continue;
        }
        cout << n*( ((n-1)*(n-2)%oo)*(n-2)%oo + (A(n-1, 2)*(n-3) %oo)*(n-3)%oo) % oo << "\n";
    }
    return 0;
}
```
