# 01. usr hbnvd - O(nlogn) - O(nlogn)
## Idea
- Giả sử $A$ được sắp xếp, ta có thể cố định $l$, sau đó có thể dùng tìm kiếm nhị phân để tìm $r$ thích hợp.

## Cpp
```cpp
#include <bits/stdc++.h>
using namespace std;

void fileIO() {
    #ifndef ONLINE_JUDGE
        freopen("testcpp1.INP", "r", stdin);
        freopen("testcpp1.OUT", "w", stdout);
    #endif
}
#define ll long long
#define fto(i, a, b) for (int i = a; i <= b; ++i)
#define ftd(i, a, b) for (int i = a; i >= b; --i)
#define fastIO() ios_base::sync_with_stdio(false); cin.tie(nullptr)

int TEST_CASE = 2;

int n;
int a[500005];
int main() {
    fileIO();
    while (TEST_CASE--) {
        cout << TEST_CASE << "-----------\n";
        
        cin >> n;
        fto(i, 1, n) cin >> a[i];
        
        sort(a+1, a+n+1);
        int ans = a[1] + a[n];
        // O(n * log2(n))
        fto(l, 1, n) {
            int low = l+1, high = n, mid;
            
            while (low <= high) {
                mid = ((low + high)>>1);
                int cur_s = a[l] + a[mid];
                
                // so sánh ans
                if (cur_s == 0) return 0;
                if (abs(cur_s) < abs(ans)) {
                    ans = cur_s;
                } else if (abs(cur_s) == abs(ans)) {
                    ans = max(ans, cur_s);
                }
                
                // điều kiện thay đổi vị trí của binary_search
                if (cur_s < 0) {
                    low = mid+1;
                } else {
                    high = mid-1;
                }
            }
        }
        cout << ans << "\n";
        
    }
    return 0;
}
```

# 02. usr hbnvd - O(nlogn) - O(nlogn)

## Idea
- Giả sử $A$ được sắp xếp, ta có thể dùng [[Two pointers]] ở 2 đầu để lùi lại từ từ.
- Mục tiêu là càng gần 0 càng tốt, nên nếu cur_s < 0 thì tăng $l$ để cur_s tăng lên. Nếu cur_s > 0 thì giảm $r$ để cur_s nhỏ lại.
- **Tại sao không lùi $l$, hay tăng $r$ ư? Tự suy nghĩ đi;)**
## Cpp
```cpp
#include <bits/stdc++.h>
using namespace std;

void fileIO() {
    #ifndef ONLINE_JUDGE
        freopen("testcpp1.INP", "r", stdin);
        freopen("testcpp1.OUT", "w", stdout);
    #endif
}
#define ll long long
#define fto(i, a, b) for (int i = a; i <= b; ++i)
#define ftd(i, a, b) for (int i = a; i >= b; --i)
#define fastIO() ios_base::sync_with_stdio(false); cin.tie(nullptr)

int TEST_CASE = 2;

int n;
int a[500005];
int main() {
    fileIO();
    while (TEST_CASE--) {
        cout << TEST_CASE << "-----------\n";
        
        cin >> n;
        fto(i, 1, n) cin >> a[i];
        
        sort(a+1, a+n+1);

        int l = 1, r = n;
        int ans = a[1] + a[n];
        
        while (l < r) {
            int cur_s = a[l] + a[r];
            
            if (cur_s == 0) return 0;
            if (abs(cur_s) < abs(ans)) {
                ans = cur_s;
            } else if (abs(cur_s) == abs(ans)) {
                ans = max(ans, cur_s);
            }

            if (cur_s < 0) ++l;
            else --r;
        }
        cout << ans << "\n";
        
    }
    return 0;
}

```
