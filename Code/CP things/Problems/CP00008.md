# 01. usr hbnvd - O(n) - O(n)
%% Bruh nó là dynamic sliding window %%
## Idea

- Tạo: $l, r = 1$ 
- Tạo: $ans=-1e9$
- Duy trì 1 biến $sum=A.sum(l, r)$ 

Khi này ta liên tục di chuyển $r$ qua phải cho tới khi $r>n$, nếu:
- $sum\le S$: không quan tâm, giữ nguyên $l$
- $sum>S$: tăng dần $l$ cho tới khi $sum\le S$ 
So sánh $ans$ với độ dài của $[l,r]$ sau khi xét xong điều kiện trên

%% 
Có thể hình dung là r liên tục tăng, l chỉ tăng để sum <= S 
r mở rộng lãnh thổ, l thu hẹp lãnh thổ
%%
## CPP
```cpp
#include <bits/stdc++.h>
using namespace std;

void fileIO() {
    #ifdef LOCAL_DEBUG
        freopen("testcpp1.INP", "r", stdin);
        freopen("testcpp1.OUT", "w", stdout);
    #endif
}
#define ll long long
#define fto(i, a, b) for (int i = a; i <= b; ++i)
#define ftd(i, a, b) for (int i = a; i >= b; --i)
#define fastIO() ios_base::sync_with_stdio(false); cin.tie(nullptr)


int n, s;
int a[100005];
int main() {
    fileIO();
    fastIO();
    
    cin >> n >> s;
    fto(i, 1, n) cin >> a[i];

	// Coreee
    int sum = 0, ans = -1e9;
    for (int l=1, r=1; r <= n; ++r) {
        sum += a[r];
        while (sum > s) {
            sum -= a[l];
            ++l;
        }
        ans = max(ans, r-l+1); // r-l+1 là công thức tính độ dài đoạn [l,r]
    }
    cout << ans << "\n";
    
    return 0;
}
```
