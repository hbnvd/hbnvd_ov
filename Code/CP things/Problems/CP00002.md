#include <bits/stdc++.h>
using namespace std;

void fileIO() {
#ifdef LOCAL_DEBUG
    freopen("TESTCPP.INP", "r", stdin);
    freopen("TESTCPP.OUT", "w", stdout);
#endif
}
#define oo 1000000007

int m, n, mem[2005][1<<10];
int m_n; // m_n = m*n

int getBit(int i, int j) {return ((i>>j)&1);}
int onBit(int i, int j) {return (i|(1<<j));}
int offBit(int i, int j) {return (i&(~(1<<j)));}

int gen(int i, int msk) {
    if (i == m_n) {
        return 1;
    }
    if (mem[i][msk] != -1) return mem[i][msk];
    int res = gen(i+1, offBit(msk, i%m));
    if (getBit(msk, i%m) == 0 && (i%m == 0 || getBit(msk, (i%m)-1) == 0)) {
        res = (res + gen(i+1, onBit(msk, i%m))) % oo;
    }
    return mem[i][msk] = res;
}
int main() {
    fileIO();

    cin >> m >> n; // xoay {n,m} => {m,n}
    m_n = m*n;
    for (int i = 1; i <= m_n; ++i) {
        for (int j = 0; j <= (1<<m); ++j) {
            mem[i][j] = -1;
        }
    }

    cout << (gen(1, 0) + gen(1, 1)) % oo << " "; 

    return 0;
}

# 02. usr Thuw - O(m\*fibo(n+2)^2) - O(m\*2^n)
- Trong độ phức tạp thời gian fibo(n+2)^2 là số mask hợp lệ, hỏi AI để biết vì tôi đếch biết giải thích;))
## Idea - dp bitmask -  bottom up
- https://marisaoj.com/submission/1141234

>Hình dung **m** là số lượng hàng sẽ dễ hiểu hơn á, chứ để 200 cột thì bit-mask đếu nào được;))

- Chọn 1 hàng rồi tạo 2 mask hợp lệ(1 mask cho hàng này, 1 cho hàng trước) cộng giữa 2 mask đó.
## Step by step
1. Tạo 1 vector masks để lưu tất cả dãy mask không có số 1 nào kề nhau
2. Tạo mảng DP \[số hàng (m)]\[mask thứ i (masks.size())]
3.  For: hàng -> mask i -> mask j + công thức truy hồi
4. Duyệt qua mảng DP để tính tổng đáp án
## CPP 
```cpp
#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;

int main() {
    int n,m;
    cin >> n >> m;
    
	// <1>
    vector<int> masks;
    for (int mask = 0; mask < (1<<n); ++mask){
        if ((mask & (mask << 1)) == 0){ // điều kiện kiểm tra mask hợp lệ
            masks.push_back(mask);
        } 
    }
    
	// <2>
    int k = (int)masks.size();
    vector<vector<long long>> dp(m + 1, vector<long long>(k, 0));
    for (int i = 0; i < k; ++i){
        dp[1][i] = 1; // đọc công thức truy hồi trước đê
    }
    
    // <3>
    for (int row = 2; row <= m; ++col){
        for (int i = 0; i < k; ++i){
            for (int j = 0; j < k; ++j){
	            // kiểm tra 2 mask có trùng số 1 nào không
                if ((masks[i] & masks[j]) == 0){
	                
	                // ct truy hồi
	                //  dp hàng thứ ROW, chọn mask là i sẽ có giá trị là...
                    dp[row][i] = (dp[row][i] + dp[row-1][j]) % mod;
                }
            }
        }
    }
	// <4>
    long long ans = 0;
    for (int i = 0; i < k; ++i){
        ans = (ans + dp[m][i]) % mod;
    }
    cout << ans << '\n';
}
```
