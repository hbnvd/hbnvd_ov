# 01. by hbnvd - O(n) - O(n)
## Idea


- Gọi hàm $f(i,c,t)$ là hàm trả về niềm vui lớn nhất nếu ta thăm người $c$ lần thứ $t$ liên tiếp vào ngày $i$

## Cpp
```cpp
int n;
int a[100005];
int b[100005];
ll mem[100005][2][3];
memset(mem, -1, sizeof(mem));
// input ...

// alice = 0, patch = 1
ll dp(int i, int c, int t) {
    if (i == 1) {
        if (c == 0) return a[1];
        return b[1];
    }
    if (mem[i][c][t] != -1) return mem[i][c][t];

    ll res = -1e18; 
    
    if (c == 0) {
	    // có thể hiểu xét dp(i-1) là xét hôm qua
        res = max(res, max(dp(i-1, Patch, 1), dp(i-1, Patch, 2)) + a[i]);
        
        if (t == 2) {
            res = max(res, dp(i-1, Alice, 1) + a[i]);
        }
    } else { 
        res = max(res, max(dp(i-1, Alice, 1), dp(i-1, Alice, 2)) + b[i]);

        if (t == 2) {
            res = max(res, dp(i-1, Patch, 1) + b[i]);
        }
    }
    return mem[i][c][t] = res;
}

cin >> n;
for (int i = 1; i <= n; ++i) {
	cin >> a[i];
}
for (int i = 1; i <= n; ++i) {
	cin >> b[i];
}
cout << max({dp(n, 0, 1), dp(n, 0, 2), dp(n, 1, 1), dp(n, 1, 2)});
```

# 02. by hbnvd - O(n) - O(n)
## Idea
- Như trên but bottom up
## Cpp
```cpp
int n;
int a[100005];
int b[100005];
ll dp[100005][2][3];
// input ...

dp[1][alice][1] = a[1];
dp[1][patch][1] = b[1];

for (int i = 1; i <= n; ++i) {
	dp[i][alice][1] = max(dp[i-1][patch][1], dp[i-1][patch][2]) + a[i];
	dp[i][patch][1] = max(dp[i-1][alice][1], dp[i-1][alice][2]) + b[i];

	dp[i][alice][2] = dp[i-1][alice][1] + a[i];
	dp[i][patch][2] = dp[i-1][patch][1] + b[i];
}

cout << max({dp[n][alice][1], dp[n][alice][2], dp[n][patch][1], dp[n][patch][2]});


```
