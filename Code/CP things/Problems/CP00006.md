# 01. usr hbnvd - O(n+m) - O(n+m)
## Idea
- Cho 2 con trỏ $i, j$ chạy qua $A, B$ , nếu:
	- $A_i < B_j$ : cho $A_i$ vào cuối $C$ , tăng $i$ thêm 1
	- $A_i > B_j$ : cho $B_j$ vào cuối $C$ , tăng $j$ thêm 1
	- $A_i = B_j$ : cho 1 trong 2 vào cuối $C$, (tùy ý), tăng cả i và j thêm 1
- Làm cho tới khi $i=n$ và $j=m$ 
- Bỏ mảng $C$, in ra số đã chọn luôn cũng được

## Cpp
```cpp
#include <bits/stdc++.h>
using namespace std;

void fileIO() {
    #ifdef LOCAL_DEBUG
        freopen("testcpp1.INP", "r", stdin);
        freopen("testcpp1.OUT", "w", stdout);
    #endif
}
#define ll long long
#define fto(i, a, b) for (int i = a; i <= b; ++i)
#define ftd(i, a, b) for (int i = a; i >= b; --i)
#define fastIO() ios_base::sync_with_stdio(false); cin.tie(nullptr)

int n, m;
int a[100005], b[100005];
int main() {
    fileIO();
    fastIO();

    cin >> n >> m;
    fto(i, 1, n) cin >> a[i];
    fto(j, 1, m) cin >> b[j];
    a[n+1] = 1e9+7; b[m+1] = 1e9+7;

    int i = 1, j = 1;
    while (i <= n || j <= m) {
        if (a[i] < b[j]) {
            cout << a[i] << " ";
            ++i;
        } else if (a[i] > b[j]) {
            cout << b[j] << " ";
            ++j;
        } else {
            cout << a[i] << " ";
            ++i; ++j;
        }
    }
    cout << "\n";
    
    
    return 0;
}
```
