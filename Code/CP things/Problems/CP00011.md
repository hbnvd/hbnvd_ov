# 01. usr hbnvd - O(n^2) - O(1)
- Brute Force
## Idea
- Ta xét ở từng cột nước riêng lẻ, ở cột thứ $A_i$  chiều ngang ($wid$) luôn là 1, độ cao ($hei$) sẽ là $\min(\text{Cột cao nhất bên trái, cột cao nhất bên phải}) - A_i$
- Khi này chỉ cần duyệt qua từng $A_i$ rồi tìm cột cao nhất bên trái, cột cao nhất bên phải rồi cộng dồn vào 1 biến tích lũy.
## Cpp
```cpp
#include <bits/stdc++.h>
using namespace std;

void fileIO() {
    #ifndef ONLINE_JUDGE
        freopen("testcpp1.INP", "r", stdin);
        freopen("testcpp1.OUT", "w", stdout);
    #endif
}
#define ll long long
#define fto(i, a, b) for (int i = a; i <= b; ++i)
#define ftd(i, a, b) for (int i = a; i >= b; --i)
#define fastIO() ios_base::sync_with_stdio(false); cin.tie(nullptr)

int TEST_CASE = 10;

int n, m;
int a[100005];
int main() {
    fileIO();
    fastIO();
    while (TEST_CASE--) {
        // cout << "-----------" << TEST_CASE << "\n";
        cin >> n;

        for (int i = 1; i <= n; ++i) {
            cin >> a[i];
        }
        
        int ans = 0;
        for (int i = 2; i <= n-1; ++i) {
            int mx_l = a[i], mx_r = a[i];

            for (int j = 1; j <= i-1; ++j) {
                mx_l = max(mx_l, a[j]);
            }
            for (int j = i+1; j <= n; ++j) {
                mx_r = max(mx_r, a[j]);
            }
            
            ans += min(mx_l, mx_r) - a[i];
        }
        cout << ans << "\n";
    }
    return 0;
}

```
# 02. usr hbnvd - O(n) - O(n)
## Idea
- Tương tự [[#01. usr hbnvd - O(n 2) - O(1)]] nhưng ta có thể thay thế việc truy vấn max (ở line 34 ~ 39) bằng prefix / suffix max.
## Cpp
```cpp
#include <bits/stdc++.h>
using namespace std;

void fileIO() {
#ifndef ONLINE_JUDGE
    freopen("testcpp1.INP", "r", stdin);
    freopen("testcpp1.OUT", "w", stdout);
#endif
}
#define ll long long
#define fto(i, a, b) for (int i = a; i <= b; ++i)
#define ftd(i, a, b) for (int i = a; i >= b; --i)
#define fastIO() ios_base::sync_with_stdio(false); cin.tie(nullptr)

int TEST_CASE = 1;

int n, m;
int a[100005];
int prf[100005];
int suf[100005];
int main() {
    fileIO();
    fastIO();
    while (TEST_CASE--) {
        // cout << "-----------" << TEST_CASE << "\n";

        cin >> n;

        for (int i = 1; i <= n; ++i) {
            cin >> a[i];
        }
        if (n <= 2) continue;

        prf[1] = a[1];
        for (int i = 2; i <= n; ++i) {
            prf[i] = max(prf[i-1], a[i]);
        }

        suf[n] = a[n];
        for (int i = n-1; i >= 1; --i) {
            suf[i] = max(suf[i+1], a[i]);
        }

        int ans = 0;
        for (int i = 1; i <= n; ++i) {
            ans += min(prf[i], suf[i]) - a[i];
        }
        cout << ans << "\n";
    }
    return 0;
}

```

# 03. hbnvd - O(n) - O(1)
## Idea

> Vách ngăn thấp nhất quyết định nước trong thùng


- Cho 2 con trỏ $l,r$ , $water$ đại diện cho tổng lượng nước trong thùng
- Cho 2 biến $mxl$ (giá trị lớn nhất bên trái $l$), $mxr$ (giá trị lớn nhất bên trái $r$), đây chính là "vách ngăn" của thùng nước mà ta cần dùng để tính.
- Lấy câu nói ở 6 dòng trên, ta sẽ ưu tiên tính lượng nước ở con trỏ có "vách ngăn" thấp hơn con trỏ còn lại.

## Cpp
```cpp
#include <bits/stdc++.h>
using namespace std;

void fileIO() {
#ifndef ONLINE_JUDGE
    freopen("testcpp1.INP", "r", stdin);
    freopen("testcpp1.OUT", "w", stdout);
#endif
}
#define ll long long
#define fto(i, a, b) for (int i = a; i <= b; ++i)
#define ftd(i, a, b) for (int i = a; i >= b; --i)
#define fastIO() ios_base::sync_with_stdio(false); cin.tie(nullptr)

int TEST_CASE = 1;

int n, m;
int a[100005];
int prf[100005];
int suf[100005];
int main() {
    fileIO();
    fastIO();
    while (TEST_CASE--) {
        // cout << "-----------" << TEST_CASE << "\n";

        cin >> n;

        for (int i = 1; i <= n; ++i) {
            cin >> a[i];
        }
        if (n <= 2) {
            cout << "0\n"; continue;
        }

        int l = 2, r = n-1; // tường sát biên không thể chứa nước
        int mxl = a[1], mxr = a[n];

        int water = 0;
        while (l <= r) {
            if (mxl < mxr) {
                water += max(0, mxl - a[l]); // Trường hợp đụng vách ngăn mới
                mxl = max(mxl, a[l]); // vừa đi vừa tính vách ngăn mới;)
                ++l;
            } else {
                water += max(0, mxr - a[r]);
                mxr = max(mxr, a[r]);
                --r;
            }
        }

        cout << water << "\n";
    }
    return 0;
}

```
