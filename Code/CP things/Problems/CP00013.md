// dùng cài này nếu đề kêu trả về idx, không thì xài hàm min
int get_best_idx(int i, int j) { 
    if (a[i] < a[j]) return i;
    if (a[j] < a[i]) return j;
    if (i < j) return i;
    return j;
}

// Init , input
#define LOG 20
ll a[maxN];
ll sp[LOG][maxN];

//

memset(sp, 0, sizeof(sp));

int n, q; cin >> n >> q;
for (int i = 1; i <= n; ++i) {
	cin >> a[i];
	sp[0][i] = i; // copy
}
// CORE
// tạo sp table
for (int i = 1; i < LOG; ++i) {
	for (int j = 1; j <= n - (1<<i) + 1; ++j) {
		sp[i][j] = get_best_idx(sp[i-1][j], sp[i-1][j+(1<<(i-1))]);
	}
}
// truy vấn
while (q--) {
	int l, r; cin >> l >> r;
	int k = __lg(r-l+1); // (int)log2(r-l+1)
	cout << get_best_idx(sp[k][l], sp[k][r-(1<<k)+1]) << "\n";

    }
